std/
├── builtins/              # Compiler intrinsics with no declaration behind them
├── core/                  # Fundamental primitives, types, and language traits
│   ├── Array              # Fixed-size primitive array and slicing
│   ├── Assert             # Basic assertion mechanisms
│   ├── Error              # Throw, Result, Failure, Fallible; attempt and rethrow
│   ├── Iter               # The Iter type that `for` consumes
│   ├── Ops                # Fundamental operator traits
│   ├── Option             # Optional value type (Some/None)
│   └── Range              # Slice and loop bounds
├── iter/                  # Building and transforming sequences
│   ├── Yield              # The Yield effect, and `iterator` to pull a producer lazily
│   └── Adapters           # map, filter, zip, enumerate, take, chain, collect over Iter
├── collections/           # Data structures and collection interfaces
│   ├── HashMap            # Hashed key-value map implementation
│   ├── HashSet            # Hashed unique-member set implementation
│   ├── List               # Growable list collection & eager sequence utilities
│   ├── Map                # Map interface / literal target
│   └── Set                # Set interface / literal target
├── text/                  # String manipulation and pattern matching
│   ├── String             # String primitives and methods
│   ├── StringBuilder      # Incremental string construction buffer
│   ├── Format             # Display and Debug, and their derives; number formatting (precision, padding, radix)
│   ├── Utf8               # Bytes to string and back; decoding that fails as a value
│   ├── Regex              # Regular expressions (AST/Parser hidden internally, built on algo/automata)
│   └── regex/             # The pattern's syntax tree, its parser, and its automaton
├── math/                  # Numeric and mathematical utilities
│   ├── Math               # min, max, abs
│   └── BigInt             # Arbitrary-precision integers, with the operator traits implemented
├── algo/                  # General-purpose algorithms
│   ├── automata/          # Finite automata over integer symbols
│   │   ├── Nfa            # Nondeterministic automaton, epsilon closure, simulation
│   │   └── Dfa            # Deterministic automaton, subset construction, minimisation
│   ├── Sort               # Stable and unstable sorts, by order, key or comparison
│   ├── Search             # Binary search, lower and upper bound, partition point
│   └── graph/             # Graphs and the algorithms over them
├── encoding/              # Serialization and data format parsers
│   ├── Codec              # Encode and Decode traits and their derives; the Encoder/Decoder interface formats implement
│   ├── Base64             # Standard and URL-safe alphabets, with and without padding
│   ├── Hex                # Bytes to hex digits and back
│   ├── Json               # JSON parser, serializer, and value AST
│   └── Toml               # TOML configuration reader and writer
├── ops/                   # Common traits and conversions
│   ├── Hash               # Hashing interface for collections
│   ├── To                 # Infallible type conversion trait
│   └── TryTo              # Fallible type conversion trait
├── meta/                  # Compile-time reflection and code generation
│   ├── Reflect            # TypeShape, fields, variants, attributes -- what typeof answers
│   ├── Name               # Identifiers, and building them from strings
│   └── Derive             # Helpers for writing derivers: walking a shape, emitting impls
├── async/                 # Cooperative tasks: block_on, all/both, promises, pause
│   ├── Task               # block_on, the all/both/interleaved scopes, pause
│   ├── Promise            # promise, resolve, await
│   ├── Channel            # Bounded and unbounded channels between tasks
│   ├── Timer              # sleep and timeout
│   └── Select             # Waiting on whichever of several is ready first; race
├── io/                    # Streams, the terminal, and printing
│   ├── Io                 # Reader, Writer, Closer, IoError; lines and using
│   ├── Buffer             # Growable in-memory bytes that are a Reader and a Writer; endian ints
│   ├── Console            # Console and Stdin effects, read_line
│   └── Print              # print and printerr, imported globally by the prelude
├── fs/                    # File system operations
│   ├── File               # File handle management and I/O
│   └── Path               # Platform-agnostic path manipulation
├── net/                   # Low-level networking primitives
├── os/                    # Operating system integration & environmental controls
│   ├── Args               # Command-line arguments iterator
│   ├── Cli                # Typed argument parsing: derive a parser from a struct, with generated help
│   ├── Env                # Environment variable management
│   ├── Process            # Subprocess spawning, control, and exit codes
│   └── Time               # Clocks, durations, instants, and system timestamps
├── test/                  # Built-in testing and assertion framework
│   ├── Test               # Test case runner and assertions
│   └── Bench              # Micro-benchmarking utilities
├── crypto/                # Cryptographic digests and primitives
│   └── Sha256             # SHA-256 digest; a Writer, so a stream can be hashed as it is read
├── random/                # Random number generation, including a secure source
└── log/                   # Structured logging infrastructure
    └── Logger             # Logging facade and severity levels
